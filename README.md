Instalacion
git clone https://github.com/nek-3231/Paty-vulgarsota-.git
cd Paty-vulgarsota-
pip install -r requirements.txt

Con Gemini
export GEMINI_API_KEY="tu-pinche-api-key"
python3 main.py archivo.py --gemini

Con Ollama Local
ollama serve &
python3 main.py archivo.py
